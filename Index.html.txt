<script>
  // Generate floating particles
  const particlesEl = document.getElementById('particles');
  for (let i = 0; i < 30; i++) {
    const p = document.createElement('div');
    p.className = 'particle';
    p.style.cssText = `
      left: ${Math.random() * 100}%;
      animation-duration: ${6 + Math.random() * 10}s;
      animation-delay: ${Math.random() * 8}s;
      width: ${1 + Math.random() * 3}px;
      height: ${1 + Math.random() * 3}px;
      background: ${['#FFD700','#FF6B1A','#ffffff','#138808'][Math.floor(Math.random()*4)]};
    `;
    particlesEl.appendChild(p);
  }

  // Target: May 4, 2026 8:00 AM IST (UTC+5:30)
  const target = new Date('2026-05-04T02:30:00Z'); // 8:00 AM IST = 2:30 AM UTC

  const daysEl = document.getElementById('days');
  const hoursEl = document.getElementById('hours');
  const minsEl = document.getElementById('minutes');
  const secsEl = document.getElementById('seconds');
  const progressFill = document.getElementById('progressFill');
  const pctEl = document.getElementById('pct');
  const statusDot = document.getElementById('statusDot');
  const statusText = document.getElementById('statusText');
  const countdownSection = document.getElementById('countdownSection');
  const finishedSection = document.getElementById('finishedSection');

  // Election day: Apr 23, 2026 for progress calc
  const electionDay = new Date('2026-04-23T00:00:00Z');
  const totalWindow = target - electionDay;

  let confettiFired = false;

  function pad(n) { return String(Math.floor(n)).padStart(2, '0'); }

  function flashNum(el, val) {
    const padded = pad(val);
    if (el.textContent !== padded) {
      el.textContent = padded;
      el.classList.remove('changed');
      void el.offsetWidth;
      el.classList.add('changed');
      setTimeout(() => el.classList.remove('changed'), 400);
    }
  }

  function fireConfetti() {
    if (confettiFired) return;
    confettiFired = true;
    const colors = ['#FF6B1A','#FFD700','#138808','#FF0000','#ffffff','#000080'];
    for (let i = 0; i < 80; i++) {
      setTimeout(() => {
        const c = document.createElement('div');
        c.className = 'confetti-piece';
        c.style.cssText = `
          left: ${Math.random() * 100}%;
          background: ${colors[Math.floor(Math.random() * colors.length)]};
          animation-duration: ${2 + Math.random() * 3}s;
          animation-delay: ${Math.random() * 2}s;
          border-radius: ${Math.random() > 0.5 ? '50%' : '2px'};
        `;
        document.body.appendChild(c);
        setTimeout(() => c.remove(), 6000);
      }, i * 30);
    }
  }

  function tick() {
    const now = new Date();
    const diff = target - now;

    // Update progress bar
    const elapsed = now - electionDay;
    const pct = Math.min(100, Math.max(0, (elapsed / totalWindow) * 100));
    progressFill.style.width = pct.toFixed(1) + '%';
    pctEl.textContent = pct.toFixed(0) + '%';

    if (diff <= 0) {
      // Results time!
      countdownSection.style.display = 'none';
      finishedSection.classList.add('show');
      statusDot.classList.add('green');
      statusText.textContent = 'Results are LIVE!';
      fireConfetti();
      return;
    }

    const totalSecs = Math.floor(diff / 1000);
    const d = Math.floor(totalSecs / 86400);
    const h = Math.floor((totalSecs % 86400) / 3600);
    const m = Math.floor((totalSecs % 3600) / 60);
    const s = totalSecs % 60;

    flashNum(daysEl, d);
    flashNum(hoursEl, h);
    flashNum(minsEl, m);
    flashNum(secsEl, s);

    // Update status
    if (d === 0) {
      statusText.textContent = 'Results day is TODAY!';
      statusDot.classList.add('green');
    } else if (d === 1) {
      statusText.textContent = 'Results TOMORROW — May 4, 2026';
    }
  }

  tick();
  setInterval(tick, 1000);
</script>
